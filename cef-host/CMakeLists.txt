cmake_minimum_required(VERSION 3.24)
project(office360_cef LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(USE_SANDBOX OFF CACHE BOOL "" FORCE)
set(CEF_RUNTIME_LIBRARY_FLAG "/MD" CACHE STRING "" FORCE)
if(NOT CEF_ROOT OR NOT EXISTS "${CEF_ROOT}/cmake/FindCEF.cmake")
  message(FATAL_ERROR "Set CEF_ROOT to the pinned CEF binary distribution")
endif()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
ADD_LOGICAL_TARGET(libcef_lib "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
add_subdirectory(${CEF_LIBCEF_DLL_WRAPPER_PATH} libcef_dll_wrapper)

add_library(office360_cef_host SHARED src/office360_cef_host.cpp)
target_include_directories(office360_cef_host PRIVATE "${CEF_INCLUDE_PATH}" include)
target_link_libraries(office360_cef_host PRIVATE libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS})
target_compile_definitions(office360_cef_host PRIVATE O360_CEF_EXPORTS NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(office360_cef_host PROPERTIES OUTPUT_NAME "office360_cef_host")

add_executable(office360_cef_subprocess WIN32 src/subprocess.cpp)
target_include_directories(office360_cef_subprocess PRIVATE "${CEF_INCLUDE_PATH}")
target_link_libraries(office360_cef_subprocess PRIVATE libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS})
target_compile_definitions(office360_cef_subprocess PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(office360_cef_subprocess PROPERTIES OUTPUT_NAME "office360-cef-subprocess")
